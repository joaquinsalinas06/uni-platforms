### Pipelining: 3 paquetes por RTT (ilustrativo: L/R = 2 ms, d_prop = 10 ms)

<Visualization viz={{ type: 'spacetime', title: 'Pipelining', steps: (() => {
  const columns = [{ id: 'e', label: 'Emisor' }, { id: 'r', label: 'Receptor' }];
  const time = { unit: 'ms', max: 30 };
  const d = (k) => ({ from: 'e', to: 'r', tStart: 2 * k, tTrans: 2, tProp: 10, label: `pkt ${k}`, kind: 'data' });
  const a = (k) => ({ from: 'r', to: 'e', tStart: 2 * k + 12, tProp: 10, label: `ACK ${k}`, kind: 'ack' });
  const seq = [d(0), d(1), d(2), a(0), a(1), a(2)];
  const notes = ['El primer bit de pkt 0 sale en t = 0; el último en t = L/R = 2 ms.', 'Sin esperar el ACK, el emisor transmite pkt 1…', '…y pkt 2: tres paquetes "en vuelo".', 'El ACK 0 vuelve: llega en t = 2 + 10 + 10 = 22 ms.', 'ACK 1.', 'ACK 2: en un RTT + L/R se enviaron 3 paquetes en vez de 1.'];
  return notes.map((note, i) => ({ note, columns, time,
    sends: seq.slice(0, i + 1).map((s, j) => ({ ...s, state: j === i ? 'active' : 'muted' })),
    spans: i >= 3 ? [{ column: 'e', tStart: 0, tEnd: 22, label: 'RTT + L/R', kind: 'rtt' }, { column: 'r', tStart: 10, tEnd: 12, label: 'L/R', kind: 'trans', state: 'marked' }] : [{ column: 'r', tStart: 10, tEnd: 12, label: 'L/R', kind: 'trans', state: 'marked' }],
  }));
})() }} />

  const snd = (base, sent, acked) => ({ role: 'sender', label: 'Emisor (GBN, N = 4)', base, size: 4, cells: row((n) => n < acked ? 'acked' : n < sent ? 'sent' : n < base + 4 ? 'usable' : 'unusable') });
